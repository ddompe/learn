@echo off
cd /d C:\Users\daniela\cafe-monthly
C:\Users\daniela\.local\bin\uv.exe run python pipeline.py >> output\task.log 2>&1

schtasks /Create /TN "space.dompe.cafe-pipeline" /SC MONTHLY /D 1 /ST 07:00 /TR "C:\Users\daniela\cafe-monthly\run_pipeline.cmd"
